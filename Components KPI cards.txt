import React from 'react';
import { ArrowUpRight, ArrowDownRight, LucideIcon } from 'lucide-react';

interface KPICardProps {
  title: string;
  value: string | number;
  change: string;
  isPositive: boolean;
  icon: LucideIcon;
  bgColor?: string;
  borderColor?: string;
  onClick?: () => void;
}

export function KPICard({
  title,
  value,
  change,
  isPositive,
  icon: IconComponent,
  bgColor = 'from-blue-50 to-blue-100',
  borderColor = 'border-blue-300',
  onClick,
}: KPICardProps) {
  return (
    <div
      onClick={onClick}
      className={`bg-gradient-to-br ${bgColor} border-2 ${borderColor} rounded-xl p-6 shadow-md hover:shadow-lg transition-all ${
        onClick ? 'cursor-pointer transform hover:scale-105' : ''
      }`}
    >
      {/* Header */}
      <div className="flex justify-between items-start mb-4">
        <h3 className="text-gray-600 font-medium text-sm">{title}</h3>
        <div className="bg-white rounded-lg p-2 shadow-sm">
          <IconComponent className="w-5 h-5 text-gray-700" />
        </div>
      </div>

      {/* Value */}
      <div className="mb-3">
        <p className="text-3xl font-bold text-gray-800">{value}</p>
      </div>

      {/* Change */}
      <div
        className={`flex items-center gap-1 text-sm font-semibold ${
          isPositive ? 'text-emerald-600' : 'text-red-600'
        }`}
      >
        {isPositive ? (
          <ArrowUpRight className="w-4 h-4" />
        ) : (
          <ArrowDownRight className="w-4 h-4" />
        )}
        <span>{change}</span>
      </div>
    </div>
  );
}

// KPI Grid Component
interface KPIGridProps {
  kpis: KPICardProps[];
  columns?: 1 | 2 | 3 | 4;
}

export function KPIGrid({ kpis, columns = 4 }: KPIGridProps) {
  const gridColsClass = {
    1: 'grid-cols-1',
    2: 'md:grid-cols-2',
    3: 'md:grid-cols-3',
    4: 'lg:grid-cols-4',
  };

  return (
    <div
      className={`grid grid-cols-1 ${gridColsClass[columns]} gap-6 mb-8`}
    >
      {kpis.map((kpi, index) => (
        <KPICard key={index} {...kpi} />
      ))}
    </div>
  );
}

export default KPICard;
