import React, { useState } from 'react';
import { ChevronDown, Wifi, WifiOff, Plus } from 'lucide-react';

interface HeaderProps {
  selectedBranch?: string;
  isOnline?: boolean;
  onBranchChange?: (branch: string) => void;
  onStatusChange?: (online: boolean) => void;
  onNewKOT?: () => void;
  branches?: string[];
}

const DEFAULT_BRANCHES = [
  'Main Branch - Shirdi',
  'Downtown Location',
  'Airport Plaza',
  'Mall Branch',
];

export function Header({
  selectedBranch = 'Main Branch - Shirdi',
  isOnline = true,
  onBranchChange,
  onStatusChange,
  onNewKOT,
  branches = DEFAULT_BRANCHES,
}: HeaderProps) {
  const [showBranchDropdown, setShowBranchDropdown] = useState(false);

  const handleBranchSelect = (branch: string) => {
    onBranchChange?.(branch);
    setShowBranchDropdown(false);
  };

  return (
    <header className="bg-white border-b border-gray-200 shadow-sm sticky top-0 z-40">
      <div className="px-8 py-4 flex items-center justify-between">
        {/* Left: Branch Selector */}
        <div className="flex items-center gap-2">
          <div className="relative">
            <button
              onClick={() => setShowBranchDropdown(!showBranchDropdown)}
              className="flex items-center gap-2 px-4 py-2 bg-gray-100 hover:bg-gray-200 rounded-lg transition-colors text-gray-700 font-medium text-sm"
            >
              📍 {selectedBranch}
              <ChevronDown
                className={`w-4 h-4 transition-transform ${
                  showBranchDropdown ? 'rotate-180' : ''
                }`}
              />
            </button>

            {/* Dropdown Menu */}
            {showBranchDropdown && (
              <div className="absolute top-full left-0 mt-2 w-64 bg-white border border-gray-200 rounded-lg shadow-lg z-50">
                <div className="p-2">
                  {branches.map((branch) => (
                    <button
                      key={branch}
                      onClick={() => handleBranchSelect(branch)}
                      className={`w-full text-left px-4 py-2 rounded-lg transition-colors ${
                        branch === selectedBranch
                          ? 'bg-[#FF6B00] text-white font-semibold'
                          : 'text-gray-700 hover:bg-gray-100'
                      }`}
                    >
                      {branch}
                    </button>
                  ))}
                </div>
              </div>
            )}
          </div>
        </div>

        {/* Right: Status & Actions */}
        <div className="flex items-center gap-6">
          {/* Status Badge */}
          <div className="flex items-center gap-2">
            <button
              onClick={() => onStatusChange?.(!isOnline)}
              className={`flex items-center gap-2 px-4 py-2 rounded-lg font-medium transition-all cursor-pointer ${
                isOnline
                  ? 'bg-emerald-100 text-emerald-700 hover:bg-emerald-200'
                  : 'bg-red-100 text-red-700 hover:bg-red-200'
              }`}
              title="Click to toggle online/offline status"
            >
              {isOnline ? (
                <Wifi className="w-4 h-4 animate-pulse" />
              ) : (
                <WifiOff className="w-4 h-4" />
              )}
              <span className="text-sm">{isOnline ? 'Online' : 'Offline'}</span>
            </button>
          </div>

          {/* New KOT Button */}
          <button
            onClick={onNewKOT}
            className="flex items-center gap-2 px-6 py-2.5 bg-[#FF6B00] hover:bg-orange-700 text-white font-semibold rounded-lg transition-all shadow-md hover:shadow-lg transform hover:scale-105 active:scale-95"
            title="Create a new Kitchen Order Ticket"
          >
            <Plus className="w-5 h-5" />
            <span className="text-sm">New KOT</span>
          </button>
        </div>
      </div>

      {/* Offline Banner */}
      {!isOnline && (
        <div className="px-8 py-2 bg-red-50 border-t border-red-200">
          <p className="text-sm text-red-700 font-medium">
            ⚠️ You are currently offline. Some features may be unavailable.
          </p>
        </div>
      )}
    </header>
  );
}

export default Header;
