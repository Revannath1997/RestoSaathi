  orders?: Order[];
  isLoading?: boolean;
  onViewAll?: () => void;
  onOrderClick?: (orderId: string) => void;
}

const STATUS_STYLES = {
  Ready: 'bg-emerald-100 text-emerald-800',
  Preparing: 'bg-amber-100 text-amber-800',
  Served: 'bg-blue-100 text-blue-800',
  Completed: 'bg-gray-100 text-gray-800',
};

export function RecentOrders({
  orders = DEFAULT_ORDERS,
  isLoading = false,
  onViewAll,
  onOrderClick,
}: RecentOrdersProps) {
  const [selectedOrder, setSelectedOrder] = useState<string | null>(null);

  const handleOrderClick = (orderId: string) => {
    setSelectedOrder(orderId);
    onOrderClick?.(orderId);
  };

  return (
    <div className="bg-white rounded-xl shadow-md border border-gray-200 overflow-hidden">
      {/* Header */}
      <div className="px-6 py-4 border-b border-gray-200">
        <h2 className="text-xl font-bold text-gray-800">Recent Orders</h2>
      </div>

      {/* Table or Loading State */}
      {isLoading ? (
        <div className="flex items-center justify-center py-12">
          <Loader2 className="w-8 h-8 text-[#FF6B00] animate-spin" />
        </div>
      ) : orders.length === 0 ? (
        <div className="py-12 text-center">
          <p className="text-gray-500 font-medium">No orders yet today</p>
        </div>
      ) : (
        <div className="overflow-x-auto">
          <table className="w-full">
            <thead>
              <tr className="bg-gray-50 border-b border-gray-200">
                <th className="px-6 py-3 text-left text-xs font-semibold text-gray-600">
                  KOT ID
                </th>
                <th className="px-6 py-3 text-left text-xs font-semibold text-gray-600">
                  Table
                </th>
                <th className="px-6 py-3 text-left text-xs font-semibold text-gray-600">
                  Items
                </th>
                <th className="px-6 py-3 text-left text-xs font-semibold text-gray-600">
                  Amount
                </th>
                <th className="px-6 py-3 text-left text-xs font-semibold text-gray-600">
                  Time
                </th>
                <th className="px-6 py-3 text-left text-xs font-semibold text-gray-600">
                  Status
                </th>
                <th className="px-6 py-3 text-center text-xs font-semibold text-gray-600">
                  Action
                </th>
              </tr>
            </thead>
            <tbody>
              {orders.map((order) => (
                <tr
                  key={order.id}
                  onClick={() => handleOrderClick(order.id)}
                  className={`border-b border-gray-100 hover:bg-gray-50 transition-colors cursor-pointer ${
                    selectedOrder === order.id ? 'bg-orange-50' : ''
                  }`}
                >
                  <td className="px-6 py-4 text-sm font-semibold text-gray-800">
                    {order.id}
                  </td>
                  <td className="px-6 py-4 text-sm text-gray-700 font-medium">
                    {order.table}
                  </td>
                  <td className="px-6 py-4 text-sm text-gray-600 line-clamp-2">
                    {order.items}
                  </td>
                  <td className="px-6 py-4 text-sm font-semibold text-gray-800">
                    {order.amount}
                  </td>
                  <td className="px-6 py-4 text-sm text-gray-600">{order.time}</td>
                  <td className="px-6 py-4">
                    <span
                      className={`inline-flex items-center px-3 py-1 rounded-full text-xs font-semibold ${
                        STATUS_STYLES[order.status]
                      }`}
                    >
                      {order.status}
                    </span>
                  </td>
                  <td className="px-6 py-4 text-center">
                    <ChevronRight className="w-4 h-4 text-gray-400 mx-auto" />
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      )}

      {/* Footer */}
      {!isLoading && orders.length > 0 && (
        <div className="px-6 py-4 border-t border-gray-200 flex justify-between items-center bg-gray-50">
          <p className="text-sm text-gray-600">
            Showing {orders.length} of 24 orders
          </p>
          <button
            onClick={onViewAll}
            className="px-4 py-2 text-[#FF6B00] font-semibold hover:bg-orange-50 rounded-lg transition-colors flex items-center gap-1"
          >
            View All Orders <ChevronRight className="w-4 h-4" />
          </button>
        </div>
      )}
    </div>
  );
}

export default RecentOrders;
